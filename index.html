* Anh cũng chẳng biết bắt đầu từ đâu nữa...

*/
NguoiYeu anh = new NguoiYeu("LÒ VĂN VIỆT");
NguoiThuong em = new NguoiThuong("TRẦN KHÁNH LINH");
// Anh thích những lúc được nói chuyện với em.
anh.muonDuocBen(em);
// Thích cả những điều nhỏ nhặt em kể mỗi ngày.
anh.langNgheEmKeChuyen();
// Có đôi khi anh hơi vụng về, không biết nói lời hay...
anh.vanMuonLamEmVui();
anh.luonNghiDenEm();
// Anh biết tình yêu không phải lúc nào cũng vui vẻ.
while (conYeuNhau) {
anh.namTayEm();
anh.langNgheEm();
anh.cungEmVuotQua();
}
// Nếu có lúc em buồn, mong em nhớ rằng anh vẫn ở đây.
anh.seCoGangTotHon();
anh.muonDiCungEmThatLau();
